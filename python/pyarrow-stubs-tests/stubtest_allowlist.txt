# Mismatches between the type stubs and the runtime that stubtest skips, one
# regex per line; see scripts/run_stubtest.py. Unused entries fail stubtest,
# so remove the entries of the mismatches you fix. Print the entries for the
# current mismatches with
#   python scripts/run_stubtest.py --generate-allowlist

# Stub-only modules. The pyarrow.__lib modules are checked against
# pyarrow.lib, which has no __all__.
pyarrow\._stubs_typing
pyarrow\.__lib\.[a-z]+\.__all__

# The __init__ of Cython extension types takes (*args, **kwargs) at runtime.
pyarrow\.__lib\.[a-z]+\.[A-Za-z_]+\.__init__
pyarrow\._compute\.(Expression|Function|FunctionRegistry|Kernel|UdfContext)\.__init__
